# bash completion for shortcut

_shortcut() {
    local cur prev words cword
    if type _init_completion &>/dev/null; then
        _init_completion || return
    else
        COMPREPLY=()
        cur="${COMP_WORDS[COMP_CWORD]}"
        prev="${COMP_WORDS[COMP_CWORD-1]}"
    fi

    local subcommands="run r new list show rm update completion help"
    local name_commands="run r show rm"

    if [ "${COMP_CWORD:-1}" -eq 1 ]; then
        COMPREPLY=( $(compgen -W "${subcommands}" -- "$cur") )
        return 0
    fi

    # What's before the cursor, leaving out flags: nothing yet, a namespace, or . (the project)
    local cmd="${COMP_WORDS[1]}" i
    local -a typed=()
    for (( i = 2; i < COMP_CWORD; i++ )); do
        [[ "${COMP_WORDS[i]}" == -* ]] || typed+=("${COMP_WORDS[i]}")
    done

    case " ${name_commands} " in
        *" ${cmd} "*)
            if [ "${#typed[@]}" -eq 0 ] && [[ "$cur" == -* ]]; then
                local flags
                case "$cmd" in
                    run|r) flags="-y" ;;
                    rm) flags="--local --namespace -y" ;;
                    show) flags="--json --field" ;;
                esac
                COMPREPLY=( $(compgen -W "${flags}" -- "$cur") )
            elif [ "${#typed[@]}" -le 1 ]; then
                local names
                names=$(shortcut list --names ${typed[0]} 2>/dev/null)
                COMPREPLY=( $(compgen -W "${names}" -- "$cur") )
            fi
            return 0
            ;;
    esac

    case "$prev" in
        completion)
            COMPREPLY=( $(compgen -W "bash zsh fish" -- "$cur") )
            return 0
            ;;
    esac
}

complete -F _shortcut shortcut
