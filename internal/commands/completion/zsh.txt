# zsh completion for shortcut (load after compinit)

_shortcut() {
    local -a subcommands
    subcommands=(run r new list show rm update completion help)

    if (( CURRENT == 2 )); then
        compadd -- $subcommands
        return
    fi

    case $words[2] in
        run|r|show|rm)
            local -a typed
            typed=(${words[3,CURRENT-1]:#-*})
            if (( $#typed == 0 )) && [[ $PREFIX == -* ]]; then
                case $words[2] in
                    run|r) compadd -- -y ;;
                    rm) compadd -- --local --namespace -y ;;
                    show) compadd -- --json --field ;;
                esac
            elif (( $#typed <= 1 )); then
                compadd -- ${(f)"$(shortcut list --names $typed[1] 2>/dev/null)"}
            fi
            ;;
        completion)
            (( CURRENT == 3 )) && compadd -- bash zsh fish
            ;;
    esac
}

compdef _shortcut shortcut
